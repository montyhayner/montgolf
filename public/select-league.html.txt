<script>
// Load nav bar
fetch("/partials/user-nav")
  .then(r => r.text())
  .then(html => document.getElementById("navContainer").innerHTML = html);

// Load leagues
fetch("/api/pending-leagues")
  .then(r => r.json())
  .then(data => {
    const container = document.getElementById("leagueButtons");
    data.leagues.forEach(l => {
      const btn = document.createElement("button");
      btn.textContent = l.league_name;
      btn.onclick = () => {
        fetch("/api/select-league", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ league_id: l.league_id })
        }).then(() => location.href = "/schedule");
      };
      container.appendChild(btn);
    });
  });
</script>
