// ----------------------------------------------------------------------------------------------
// Produce the CURRENT TIMESTAMP in yyyy-mm-dd hh:mn:ss 
// format and return to calling code.
// ----------------------------------------------------------------------------------------------
function formatSqliteTimestamp(d) {
  const pad = n => n.toString().padStart(2, "0");
  return (
    d.getFullYear() + "-" +
    pad(d.getMonth() + 1) + "-" +
    pad(d.getDate()) + " " +
    pad(d.getHours()) + ":" +
    pad(d.getMinutes()) + ":" +
    pad(d.getSeconds())
  );
}
