-- 1. Backup existing super_admins data
CREATE TABLE super_admins_backup AS
SELECT email
FROM super_admins;

-- 2. Drop the old table
DROP TABLE super_admins;

-- 3. Recreate the simplified table (no password_hash)
CREATE TABLE super_admins (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email TEXT NOT NULL UNIQUE
);

-- 4. Restore the data
INSERT INTO super_admins (email)
SELECT email
FROM super_admins_backup;

-- 5. Drop the temporary backup table
DROP TABLE super_admins_backup;
